def threeSum(nums):
    nums.sort()
    res = []
    for i in range(len(nums)):
        if i > 0 and nums[i] == nums[i-1]: continue
        left, right = i + 1, len(nums) - 1
        while left < right:
            s = nums[i] + nums[left] + nums[right]
            if s == 0:
                res.append([nums[i], nums[left], nums[right]])
                left += 1; right -= 1
            elif s < 0: left += 1
            else: right -= 1
    return res
📌 Invariant Guarantee:
For sorted nums, fixing nums[i] reduces problem to Two Sum II. If sum == 0, record triplet and increment L while skipping duplicates.
🎯 Active Pointers: step: 1 status: Active
✍️ Watch out for identical adjacent values! Always advance L and decrement R while nums[L] == nums[L-1] to prevent duplicate triplets.
✨ Key Takeaway: Sorting in O(N log N) transforms the cubic O(N^3) brute force search into optimal O(N^2) using bidirectional two-pointer convergence.

def trap(height):
    if not height: return 0
    left, right = 0, len(height) - 1
    left_max, right_max = 0, 0
    wSum = 0
    while left < right:
        if height[left] < height[right]:
            if height[left] >= left_max: left_max = height[left]
            else: wSum += left_max - height[left]
            left += 1
        else:
            if height[right] >= right_max: right_max = height[right]
            else: wSum += right_max - height[right]
            right -= 1
    return wSum
📌 Invariant Guarantee:
Water trapped at index i is strictly determined by min(max_left, max_right) - height[i]. The smaller boundary controls the water level.
🎯 Active Pointers: step: 1 status: Active
✍️ No need for prefix/suffix max arrays! The two-pointer approach eliminates the O(N) auxiliary space requirement entirely.
✨ Key Takeaway: Two pointers converge inward in O(N) time and O(1) space by advancing the side with the smaller max height boundary.

def canFinish(numCourses: int, prerequisites: List[List[int]]) -> bool:
    in_degree = [0] * numCourses
    adj = collections.defaultdict(list)
    for dest, src in prerequisites:
        adj[src].append(dest)
        in_degree[dest] += 1
    queue = collections.deque([i for i in range(numCourses) if in_degree[i] == 0])
    completed = []
    while queue:
        node = queue.popleft()
        completed.append(node)
        for neighbor in adj[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    return len(completed) == numCourses
📌 Invariant Guarantee:
A directed graph is a Directed Acyclic Graph (DAG) iff Kahn's BFS queue can process all V vertices. Any remaining in-degree > 0 implies a dependency cycle.
🎯 Active Pointers: step: 1 status: Active
✍️ If processed count < numCourses at termination, return false immediately — a circular dependency deadlock exists!
✨ Key Takeaway: Kahn's BFS processes nodes in O(V + E) time. Ideal for resolving task schedules, build dependency trees, and package managers.

class LRUCache {
  constructor(capacity) {
    this.capacity = capacity;
    this.map = new Map();
    this.head = null;
    this.tail = null;
  }
  put(key, value) {
    if (this.map.has(key)) {
      let node = this.map.get(key);
      node.val = value;
      this.moveToHead(node);
    } else {
      if (this.map.size >= this.capacity) {
        let lru = this.tail;
        this.removeNode(lru);
        this.map.delete(lru.key);
      }
      let newNode = { key, val: value, prev: null, next: null };
      this.addToHead(newNode);
      this.map.set(key, newNode);
    }
  }
  get(key) {
    if (!this.map.has(key)) return -1;
    let node = this.map.get(key);
    this.moveToHead(node);
    return node.val;
  }
}
📌 Invariant Guarantee:
Doubly linked list maintains chronological recency invariant (Head = MRU, Tail = LRU). Hash map maintains O(1) lookup to arbitrary nodes.
🎯 Active Pointers: step: 1 status: Active
✍️ Always use dummy sentinel Head and Tail nodes! They eliminate null-pointer boundary checks during node detachment and re-insertion.
✨ Key Takeaway: Pairing a doubly linked list with a hash map provides O(1) get, put, and node eviction by decoupling ordering from key lookup.
